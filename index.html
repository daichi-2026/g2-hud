// Speech-to-text client using OpenAI Whisper API for G2 microphone. export interface SttSnapshot { finalText: string interimText: string finished: boolean } export interface SttClient { sendPcm(chunk: Uint8Array): void close(): void flushAndTranscribe(): Promise } export function startSttStream( apiKey: string, onSnapshot: (snap: SttSnapshot) => void, onError?: (err: unknown) => void, ): SttClient { let pcmBuffer: Uint8Array[] = [] let totalLength = 0 let isClosed = false let isProcessing = false // 16kHz, 16-bit, mono PCM を WAVファイル形式（Blob）に変換するヘルパー function pcmToWavBlob(pcmChunks: Uint8Array[], totalLen: number): Blob { const wavHeader = new ArrayBuffer(44) const view = new DataView(wavHeader) const sampleRate = 16000 const channels = 1 const bitsPerSample = 16 // "RIFF" chunk descriptor view.setUint32(0, 0x52494646, false) // "RIFF" view.setUint32(4, 36 + totalLen, true) // file length - 8 view.setUint32(8, 0x57415645, false) // "WAVE" // "fmt " sub-chunk view.setUint32(12, 0x666d7420, false) // "fmt " view.setUint32(16, 16, true) // SubChunk1Size (16 for PCM) view.setUint16(20, 1, true) // AudioFormat (1 for PCM) view.setUint16(22, channels, true) view.setUint32(24, sampleRate, true) view.setUint32(28, sampleRate * channels * (bitsPerSample / 8), true) // ByteRate view.setUint16(32, channels * (bitsPerSample / 8), true) // BlockAlign view.setUint16(34, bitsPerSample, true) // "data" sub-chunk view.setUint32(36, 0x64617461, false) // "data" view.setUint32(40, totalLen, true) const wavBytes = new Uint8Array(44 + totalLen) wavBytes.set(new Uint8Array(wavHeader), 0) let offset = 44 for (const chunk of pcmChunks) { wavBytes.set(chunk, offset) offset += chunk.length } return new Blob([wavBytes], { type: 'audio/wav' }) } async function flushAndTranscribe() { if (isProcessing || pcmBuffer.length === 0 || !apiKey) return isProcessing = true const chunksToProcess = [...pcmBuffer] const lenToProcess = totalLength pcmBuffer = [] totalLength = 0 try { onSnapshot({ finalText: '', interimText: 'Translating via Whisper...', finished: false }) const wavBlob = pcmToWavBlob(chunksToProcess, lenToProcess) const formData = new FormData() formData.append('file', wavBlob, 'audio.wav') formData.append('model', 'whisper-1') formData.append('language', 'en') // 必要に応じて変更 const response = await fetch('https://api.openai.com/v1/audio/transcriptions', { method: 'POST', headers: { 'Authorization': `Bearer ${apiKey}`, }, body: formData, }) if (!response.ok) { throw new Error(`OpenAI API error: ${response.statusText}`) } const data = await response.json() const transcript = data.text || '' onSnapshot({ finalText: transcript, interimText: '', finished: true, }) } catch (err) { if (onError) onError(err) } finally { isProcessing = false } } return { sendPcm(chunk: Uint8Array) { if (isClosed) return pcmBuffer.push(chunk) totalLength += chunk.length // 例: 3秒〜5秒相当のデータが溜まったら自動で切り出して音声認識に投げる（画面1: 自動ハンズフリー用） // 16kHz * 2bytes/sample = 32,000 bytes/sec if (totalLength > 32000 * 4) { flushAndTranscribe() } }, async flushAndTranscribe() { await flushAndTranscribe() }, close() { isClosed = true pcmBuffer = [] totalLength = 0 }, } }
